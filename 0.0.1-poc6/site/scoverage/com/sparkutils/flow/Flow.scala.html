169   private def performStep(df: DataFrame, step: Step[RP], dependencies: Set[StepResult[RP]]): StepResult[RP] = {
170     // ensure the same session is used for the df, otherwise the session may fall back to the classic when called
171     // from another thread, mostly a testing issue, but would also apply to DBR using connect on a classic cluster
172     SparkSession.setActiveSession(df.sparkSession)
173     try {
174       val actualStep = modifyStep(df, step, dependencies)
175 
176       val vl = loadViews(df.sparkSession, actualStep)
177       loadMaps(df.sparkSession, actualStep)
178       stepViewsLoaded(vl, actualStep)
179       val starter = startStep(df, actualStep, dependencies)
180       val ((res, processed), runner) = process(starter, actualStep)
181 
182       val finalDF = stepCompleted(res, actualStep, dependencies)
183       // allow early exit
184       earlyExitCheck(finalDF, step, dependencies)
185 
186       if (isTraceEnabled() || showInterim) {
187         // $COVERAGE-OFF$
188         infoLogStep(actualStep, "Result Sample")
189         finalDF.show()
190         // $COVERAGE-ON$
191       }
192 
193       val timings = StepTimings(runner - processed, processed)
194 
195       infoLogStep(step, s"runner took ${timings.runner.pretty} with result processing taking ${timings.result.pretty}")
196       StepResult(actualStep, finalDF, timings)
197     } catch {
198       case f: FlowException =>
199         errorLogStep(step, f.msg)
200         throw f
201       case t: Throwable =>
202         // $COVERAGE-OFF$
203         val info = s"An unexpected error occurred during processing the Step - ${t.getMessage}"
204         errorLogStep(step, info)
205         // $COVERAGE-ON$
206         throw FlowException(flowInfo(step, info), t)
207     }
208   }

211    * Builds the chain of Promises from roots, completing roots triggers processing of the rest of the DAG
212    * @return (roots, full) promises

214   private def buildPromises(): (Map[String, Promise[StepResult[RP]]], Map[String, Promise[StepResult[RP]]]) = {
215     val rootPromises = roots.map(s => s.name -> Promise[StepResult[RP]]()).toMap
216     val promises = scala.collection.mutable.Map.empty ++ rootPromises
217 
218     def chaseDown(elem: theGraph.NodeT): Future[StepResult[RP]] = {
219       val cur = elem.source
220 
221       val r =
222         promises.get(cur.name).map(_.future).getOrElse {
223           val p = Promise[StepResult[RP]]()
224           promises.put(cur.name, p)
225 
226           // do their roots
227           val f = Future.sequence(
228             elem.diPredecessors.map {
229               parent =>
230                 chaseDown(parent)
231             }
232           )
233 
234           val newF: Future[StepResult[RP]] = f.map { names =>
235             performStep(names.head.output, cur, names)
236           }
237 
238           p.completeWith(newF)
239           p.future
240         }
241 
242       // children should stitch on their parent only
243       elem.diSuccessors.foreach {
244         dependent =>
245           chaseDown(dependent)
246       }
247 
248       r
249     }
250     rest.foreach{r => chaseDown(theGraph.get(r))}
251 
252     (rootPromises, scala.collection.immutable.Map.empty ++ promises)
253   }

258   def verifySteps(): Unit = {
259     def badViewName(name: String): Boolean = {
260       name.isEmpty
261     }
262     try {
263       if (steps.isEmpty) {
264         throw FlowException(EmptyFlow)
265       }
266       if (steps.exists(s => (s.name eq null) || s.name.isEmpty)) {
267         throw FlowException(EmptyStepName)
268       }
269       steps.find(s => (s.data.inputView.exists(badViewName) || s.data.inputView.isEmpty) && s.dependencies.size > 1).foreach {
270         step =>
271           throw FlowException(DefaultViewNamesMultipleParents(step))
272       }
273       steps.find(s => s.data.inputView.exists(badViewName) || s.data.outputView.exists(badViewName)).foreach {
274         step =>
275           throw FlowException(InvalidViewNames(step))
276       }
277       val dupes = steps.groupBy(_.name).filter(_._2.size > 1)
278       if (dupes.nonEmpty) {
279         throw FlowException(s"$DuplicateNames (${dupes.keys.mkString(",")})")
280       }
281       val cyc = theGraph.findCycle
282       // for info only val paths = roots.map( n => n -> theGraph.get(n).outerNodeTraverser.toSeq )
283       cyc.foreach {
284         cycle =>
285           throw FlowException(s"$CycleDetected $cycle")
286       }
287 
288       steps.foreach {
289         step =>
290           (step.operation.function, step.operation.resultApproach) match {
291             case (DQ, MergeFields | OutputFieldsOnly) =>
292               throw FlowException(InvalidDQResultApproach(step))
293             case _ => ()
294           }
295       }
296     } catch {
297       case f: FlowException =>
298         logError(f.msg, f)
299         throw f
300     }
301   }

304    * A convenience function to start processing with an optionally provided DataFrame registered as the first steps inputView.
305    *
306    * The entire Flow must be complete within the Flow.duration
307    *
308    * @param sparkSession
309    * @param starting a function called for each Root Step (a Step having no dependencies), when a DataFrame is provided
310    *                 it is registered as a temporary view with the name of the inputView.  When None is returned and the
311    *                 inputView is not present an error is thrown, this is delegated to the FlowDataHandling.loadData function
312    * @return a map of step name to tuples of end Steps with their resulting DataFrames and run statistics,
313    *         note although the Step returned may have been replaced by modifyStep
314    *         the name key will be the original [[Step.name]]
315    */
316   def run(sparkSession: SparkSession, starting: Step[RP] => Option[DataFrame]): FlowResult[RP] = {
317     val timed = Timer {
318       verifySteps()
319       logInfo(s"Starting Flow id: ${flowId.id}, version: ${flowId.version}")
320 
321       val (rootPromises, promises) = buildPromises()
322 
323       // processing th roots sets of the rest
324       roots.par.map {
325         root =>
326           val p = rootPromises(root.name)
327           try {
328             val df = starting(root)
329             val token = rootToken(root)
330             val r =
331               df.fold(
332                 performStep(loadData(sparkSession = sparkSession, token = token), root, Set.empty)
333               ) { df =>
334                 df.createOrReplaceTempView(token)
335                 performStep(df, root, Set.empty)
336               }
337 
338             p success r
339           } catch {
340             case t: Throwable => p failure t
341           }
342       }
343 
344       val all = Future.sequence((rootPromises ++ promises).map(_._2.future))
345       val r = Await.result(all, duration)
346 
347       r.map(p => p.step.name -> p).toMap
348     }
349     logInfo(s"Finished Flow id: ${flowId.id}, version: ${flowId.version} in ${timed._2.pretty}")
350 
351     FlowResult(timed._1, timed._2)
352   }

380       logError(err.msg, err)
381       throw err
382     }
383     if (vl.notLoadedViews.nonEmpty) {
384       if (vl.notLoadedViews.exists(step.data.inputView.contains)) {
385         val err = FlowException(s"View Cycle - Step ${step.name}, Flow id: ${flowId.id}, version: ${flowId.version} could not be started as the Steps inputView `${step.data.inputView}` could not be loaded")
386         logError(err.msg, err)
387         throw err
388       } else {
389         val msg = s"View Cycle - Step ${step.name}, Flow id: ${flowId.id}, version: ${flowId.version} could not load the following views - "
390         logWarning(msg + vl.notLoadedViews.map(v => s"`$v`").mkString)
391       }
392     } else {
393       infoLogStep(step, "Views Loaded")
394     }
