275   private def performStep(df: DataFrame, step: Step[RP], dependencies: Set[StepResult[RP]]): StepResult[RP] = {
276     // ensure the same session is used for the df, otherwise the session may fall back to the classic when called
277     // from another thread, mostly a testing issue, but would also apply to DBR using connect on a classic cluster
278     SparkSession.setActiveSession(df.sparkSession)
279     try {
280       val actualStep = modifyStep(df, step, dependencies)
281 
282       val vl = loadViews(df.sparkSession, actualStep)
283       loadMaps(df.sparkSession, actualStep)
284       stepViewsLoaded(vl, actualStep)
285       val starter = startStep(df, actualStep, dependencies)
286       val ((res, processed), runner) = process(starter, actualStep)
287 
288       val finalDF = stepCompleted(res, actualStep, dependencies)
289       // allow early exit
290       earlyExitCheck(finalDF, step, dependencies)
291 
292       if (isTraceEnabled() || showInterim) {
293         // $COVERAGE-OFF$
294         infoLogStep(actualStep, "Result Sample")
295         finalDF.show()
296         // $COVERAGE-ON$
297       }
298 
299       val timings = StepTimings(runner - processed, processed)
300 
301       infoLogStep(step, s"runner took ${timings.runner.pretty} with result processing taking ${timings.result.pretty}")
302       StepResult(actualStep, finalDF, timings)
303     } catch {
304       case f: FlowException =>
305         errorLogStep(step, f.msg)
306         throw f
307       case t: Throwable =>
308         // $COVERAGE-OFF$
309         val info = s"An unexpected error occurred during processing the Step - ${t.getMessage}"
310         errorLogStep(step, info)
311         // $COVERAGE-ON$

317    * Builds the chain of Promises from roots, completing roots triggers processing of the rest of the DAG
318    * @return (roots, full) promises
319    */
320   private def buildPromises(): (Map[String, Promise[StepResult[RP]]], Map[String, Promise[StepResult[RP]]]) = {
321     val rootPromises = roots.map(s => s.name -> Promise[StepResult[RP]]()).toMap
322     val promises = scala.collection.mutable.Map.empty ++ rootPromises
323 
324     def chaseDown(elem: theGraph.NodeT): Future[StepResult[RP]] = {
325       val cur = elem.source
326 
327       val r =
328         promises.get(cur.name).map(_.future).getOrElse {
329           val p = Promise[StepResult[RP]]()
330           promises.put(cur.name, p)
331 
332           // do their roots
333           val f = Future.sequence(
334             elem.diPredecessors.map {
335               parent =>
336                 chaseDown(parent)
337             }
338           )
339 
340           val newF: Future[StepResult[RP]] = f.map { names =>
341             performStep(names.head.output, cur, names)
342           }
343 
344           p.completeWith(newF)
345           p.future
346         }
347 
348       // children should stitch on their parent only
349       elem.diSuccessors.foreach {
350         dependent =>
351           chaseDown(dependent)
352       }
353 
354       r
355     }
356     rest.foreach{r => chaseDown(theGraph.get(r))}
357 
358     (rootPromises, scala.collection.immutable.Map.empty ++ promises)
359   }

364   def verifySteps(): Unit = {
365     def badViewName(name: String): Boolean = {
366       name.isEmpty
367     }
368     try {
369       if (steps.isEmpty) {
370         throw FlowException(EmptyFlow)
371       }
372       if (steps.exists(s => (s.name eq null) || s.name.isEmpty)) {
373         throw FlowException(EmptyStepName)
374       }
375       steps.find(s => (s.data.inputView.exists(badViewName) || s.data.inputView.isEmpty) && s.dependencies.size > 1).foreach {
376         step =>
377           throw FlowException(DefaultViewNamesMultipleParents(step))
378       }
379       steps.find(s => s.data.inputView.exists(badViewName) || s.data.outputView.exists(badViewName)).foreach {
380         step =>
381           throw FlowException(InvalidViewNames(step))
382       }
383       val dupes = steps.groupBy(_.name).filter(_._2.size > 1)
384       if (dupes.nonEmpty) {
385         throw FlowException(s"$DuplicateNames (${dupes.keys.mkString(",")})")
386       }
387       val cyc = theGraph.findCycle
388       // for info only val paths = roots.map( n => n -> theGraph.get(n).outerNodeTraverser.toSeq )
389       cyc.foreach {
390         cycle =>
391           throw FlowException(s"$CycleDetected $cycle")
392       }
393 
394       steps.foreach {
395         step =>
396           (step.operation.function, step.operation.resultApproach) match {
397             case (DQ, MergeFields | OutputFieldsOnly) =>
398               throw FlowException(InvalidDQResultApproach(step))
399             case _ => ()
400           }
401       }
402     } catch {
403       case f: FlowException =>
404         logError(f.msg, f)
405         throw f
406     }
407   }

411    *
412    * The entire Flow must be complete within the Flow.duration

414    * @param sparkSession
415    * @param starting a function called for each Root Step (a Step having no dependencies), when a DataFrame is provided

417    *                 inputView is not present an error is thrown, this is delegated to the FlowDataHandling.loadData function
418    * @return a map of step name to tuples of end Steps with their resulting DataFrames and run statistics,
419    *         note although the Step returned may have been replaced by modifyStep
420    *         the name key will be the original [[Step.name]]

422   def run(sparkSession: SparkSession, starting: Step[RP] => Option[DataFrame]): FlowResult[RP] = {
423     val timed = Timer {
424       verifySteps()
425       logInfo(s"Starting Flow id: ${flowId.id}, version: ${flowId.version}")
426 
427       val (rootPromises, promises) = buildPromises()
428 
429       // processing th roots sets of the rest
430       roots.par.map {
431         root =>
432           val p = rootPromises(root.name)
433           try {
434             val df = starting(root)
435             val token = rootToken(root)
436             val r =
437               df.fold(
438                 performStep(loadData(sparkSession = sparkSession, token = token), root, Set.empty)
439               ) { df =>
440                 df.createOrReplaceTempView(token)
441                 performStep(df, root, Set.empty)
442               }
443 
444             p success r
445           } catch {
446             case t: Throwable => p failure t
447           }
448       }
449 
450       val all = Future.sequence((rootPromises ++ promises).map(_._2.future))
451       val r = Await.result(all, duration)
452 
453       r.map(p => p.step.name -> p).toMap
454     }
455     logInfo(s"Finished Flow id: ${flowId.id}, version: ${flowId.version} in ${timed._2.pretty}")
456 
457     FlowResult(timed._1, timed._2)
