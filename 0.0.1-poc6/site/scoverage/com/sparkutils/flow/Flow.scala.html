212   private def buildPromises(): (Map[String, Promise[StepResult[RP]]], Map[String, Promise[StepResult[RP]]]) = {
213     val rootPromises = roots.map(s => s.name -> Promise[StepResult[RP]]()).toMap
214     val promises = scala.collection.mutable.Map.empty ++ rootPromises
215 
216     def chaseDown(elem: theGraph.NodeT): Future[StepResult[RP]] = {
217       val cur = elem.source
218 
219       val r =
220         promises.get(cur.name).map(_.future).getOrElse {
221           val p = Promise[StepResult[RP]]()
222           promises.put(cur.name, p)
223 
224           // do their roots
225           val f = Future.sequence(
226             elem.diPredecessors.map {
227               parent =>
228                 chaseDown(parent)
229             }
230           )
231 
232           stepTimeout(cur, p)
233 
234           val newF: Future[StepResult[RP]] = f.map { names =>
235             performStep(names.head.output, cur, names)
236           }
237 
238           p.completeWith(newF)
239 
240           p.future
241         }
242 
243       // children should stitch on their parent only
244       elem.diSuccessors.foreach {
245         dependent =>
246           chaseDown(dependent)
247       }
248 
249       r
250     }
251     rest.foreach{r => chaseDown(theGraph.get(r))}
252 
253     (rootPromises, scala.collection.immutable.Map.empty ++ promises)
254   }

259   def verifySteps(): Unit = {
260     def badViewName(name: String): Boolean = {
261       name.isEmpty
262     }
263     try {
264       if (steps.isEmpty) {
265         throw FlowException(EmptyFlow)
266       }
267       if (steps.exists(s => (s.name eq null) || s.name.isEmpty)) {
268         throw FlowException(EmptyStepName)
269       }
270       steps.find(s => (s.data.inputView.exists(badViewName) || s.data.inputView.isEmpty) && s.dependencies.size > 1).foreach {
271         step =>
272           throw FlowException(DefaultViewNamesMultipleParents(step))
273       }
274       steps.find(s => s.data.inputView.exists(badViewName) || s.data.outputView.exists(badViewName)).foreach {
275         step =>
276           throw FlowException(InvalidViewNames(step))
277       }
278       val dupes = steps.groupBy(_.name).filter(_._2.size > 1)
279       if (dupes.nonEmpty) {
280         throw FlowException(s"$DuplicateNames (${dupes.keys.mkString(",")})")
281       }
282       val cyc = theGraph.findCycle
283       // for info only val paths = roots.map( n => n -> theGraph.get(n).outerNodeTraverser.toSeq )
284       cyc.foreach {
285         cycle =>
286           throw FlowException(s"$CycleDetected $cycle")
287       }
288 
289       steps.foreach {
290         step =>
291           (step.operation.function, step.operation.resultApproach) match {
292             case (DQ, MergeFields | OutputFieldsOnly) =>
293               throw FlowException(InvalidDQResultApproach(step))
294             case _ => ()
295           }
296       }
297     } catch {
298       case f: FlowException =>
299         logError(f.msg, f)
300         throw f
301     }

306    *
307    * The entire Flow must be complete within the Flow.duration

313    * @param tolerant defaulting to false, an exception in an Step triggers an Exception from this function, a successful

315    * @return a map of step name to tuples of end Steps with their resulting DataFrames and run statistics,

317    *         the name key will be the original [[Step.name]]

319   def run(sparkSession: SparkSession, starting: Step[RP] => Option[DataFrame], tolerant: Boolean = false): FlowResult[RP] = {
320     val timed = Timer {
321       verifySteps()
322       logInfo(s"Starting Flow id: ${flowId.id}, version: ${flowId.version}")
323 
324       val (rootPromises, promises) = buildPromises()
325 
326       // processing th roots sets of the rest
327       roots.par.map {
328         root =>
329           val p = rootPromises(root.name)
330           try {
331             val df = starting(root)
332             val token = rootToken(root)
333 
334             stepTimeout(root, p)
335 
336             val r =
337               df.fold(
338                 performStep(loadData(sparkSession = sparkSession, token = token), root, Set.empty)
339               ) { df =>
340                 df.createOrReplaceTempView(token)
341                 performStep(df, root, Set.empty)
342               }
343 
344             p success r
345           } catch {
346             case t: Throwable => p tryFailure StepException(root, t) // wrap this in an exception with the root
347           }
348       }
349 
350       if (tolerant) {
351         val all = Future.sequence((rootPromises ++ promises).map(_._2.future.transform(Success(_))))
352         val r = Await.result(all, duration)
353 
354         r.map {
355           case Success(stepResult: StepResult[RP]) => stepResult.step.name -> stepResult
356           case Failure(stepException: StepException[RP]) => stepException.step.name -> stepException
357         }.toMap
358       } else {
359         val all = Future.sequence((rootPromises ++ promises).map(_._2.future))
360         val r = Await.result(all, duration)
361 
362         r.map{stepResult => stepResult.step.name -> stepResult}.toMap
363       }
364     }
365     logInfo(s"Finished Flow id: ${flowId.id}, version: ${flowId.version} in ${timed._2.pretty}")
366 
367     FlowResult(timed._1, timed._2)
368   }

414       logError(err.msg, err)

416     }
417     if (vl.notLoadedViews.nonEmpty) {

419         val err = FlowException(s"View Cycle - Step ${step.name}, Flow id: ${flowId.id}, version: ${flowId.version} could not be started as the Steps inputView `${step.data.inputView}` could not be loaded")
420         logError(err.msg, err)

422       } else {
423         val msg = s"View Cycle - Step ${step.name}, Flow id: ${flowId.id}, version: ${flowId.version} could not load the following views - "
424         logWarning(msg + vl.notLoadedViews.map(v => s"`$v`").mkString)
425       }
426     } else {
427       infoLogStep(step, "Views Loaded")
428     }
