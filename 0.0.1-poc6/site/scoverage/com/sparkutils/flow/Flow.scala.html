209    * Builds the chain of Promises from roots, completing roots triggers processing of the rest of the DAG
210    * @return (roots, full) promises

212   private def buildPromises(): (Map[String, Promise[StepResult[RP]]], Map[String, Promise[StepResult[RP]]]) = {
213     val rootPromises = roots.map(s => s.name -> Promise[StepResult[RP]]()).toMap
214     val promises = scala.collection.mutable.Map.empty ++ rootPromises
215 
216     def chaseDown(elem: theGraph.NodeT): Future[StepResult[RP]] = {
217       val cur = elem.source
218 
219       val r =
220         promises.get(cur.name).map(_.future).getOrElse {
221           val p = Promise[StepResult[RP]]()
222           promises.put(cur.name, p)
223 
224           // do their roots
225           val f = Future.sequence(
226             elem.diPredecessors.map {
227               parent =>
228                 chaseDown(parent)
229             }
230           )
231 
232           val newF: Future[StepResult[RP]] = f.map { names =>
233             performStep(names.head.output, cur, names)
234           }
235 
236           p.completeWith(newF)
237           p.future
238         }
239 
240       // children should stitch on their parent only
241       elem.diSuccessors.foreach {
242         dependent =>
243           chaseDown(dependent)
244       }
245 
246       r
247     }
248     rest.foreach{r => chaseDown(theGraph.get(r))}
249 
250     (rootPromises, scala.collection.immutable.Map.empty ++ promises)
251   }

256   def verifySteps(): Unit = {
257     def badViewName(name: String): Boolean = {
258       name.isEmpty
259     }
260     try {
261       if (steps.isEmpty) {
262         throw FlowException(EmptyFlow)
263       }
264       if (steps.exists(s => (s.name eq null) || s.name.isEmpty)) {
265         throw FlowException(EmptyStepName)
266       }
267       steps.find(s => (s.data.inputView.exists(badViewName) || s.data.inputView.isEmpty) && s.dependencies.size > 1).foreach {
268         step =>
269           throw FlowException(DefaultViewNamesMultipleParents(step))
270       }
271       steps.find(s => s.data.inputView.exists(badViewName) || s.data.outputView.exists(badViewName)).foreach {
272         step =>
273           throw FlowException(InvalidViewNames(step))
274       }
275       val dupes = steps.groupBy(_.name).filter(_._2.size > 1)
276       if (dupes.nonEmpty) {
277         throw FlowException(s"$DuplicateNames (${dupes.keys.mkString(",")})")
278       }
279       val cyc = theGraph.findCycle
280       // for info only val paths = roots.map( n => n -> theGraph.get(n).outerNodeTraverser.toSeq )
281       cyc.foreach {
282         cycle =>
283           throw FlowException(s"$CycleDetected $cycle")
284       }
285 
286       steps.foreach {
287         step =>
288           (step.operation.function, step.operation.resultApproach) match {
289             case (DQ, MergeFields | OutputFieldsOnly) =>
290               throw FlowException(InvalidDQResultApproach(step))
291             case _ => ()
292           }
293       }
294     } catch {
295       case f: FlowException =>
296         logError(f.msg, f)
297         throw f
298     }

302    * A convenience function to start processing with an optionally provided DataFrame registered as the first steps inputView.
303    *
304    * The entire Flow must be complete within the Flow.duration
305    *
306    * @param sparkSession
307    * @param starting a function called for each Root Step (a Step having no dependencies), when a DataFrame is provided
308    *                 it is registered as a temporary view with the name of the inputView.  When None is returned and the
309    *                 inputView is not present an error is thrown, this is delegated to the FlowDataHandling.loadData function
310    * @param tolerant defaulting to false, an exception in an Step triggers an Exception from this function, a successful
311    *                 result will only have [[StepResult]]s.  Providing true will capture both successes and failures
312    * @return a map of step name to tuples of end Steps with their resulting DataFrames and run statistics,
313    *         note although the Step returned may have been replaced by modifyStep
314    *         the name key will be the original [[Step.name]]

316   def run(sparkSession: SparkSession, starting: Step[RP] => Option[DataFrame], tolerant: Boolean = false): FlowResult[RP] = {
317     val timed = Timer {
318       verifySteps()
319       logInfo(s"Starting Flow id: ${flowId.id}, version: ${flowId.version}")
320 
321       val (rootPromises, promises) = buildPromises()
322 
323       // processing th roots sets of the rest
324       roots.par.map {
325         root =>
326           val p = rootPromises(root.name)
327           try {
328             val df = starting(root)
329             val token = rootToken(root)
330             val r =
331               df.fold(
332                 performStep(loadData(sparkSession = sparkSession, token = token), root, Set.empty)
333               ) { df =>
334                 df.createOrReplaceTempView(token)
335                 performStep(df, root, Set.empty)
336               }
337 
338             p success r
339           } catch {
340             case t: Throwable => p failure StepException(root, t) // wrap this in an exception with the root
341           }
342       }
343 
344       if (tolerant) {
345         val all = Future.sequence((rootPromises ++ promises).map(_._2.future.transform(Success(_))))
346         val r = Await.result(all, duration)
347 
348         r.map {
349           case Success(stepResult: StepResult[RP]) => stepResult.step.name -> stepResult
350           case Failure(stepException: StepException[RP]) => stepException.step.name -> stepException
351         }.toMap
352       } else {
353         val all = Future.sequence((rootPromises ++ promises).map(_._2.future))
354         val r = Await.result(all, duration)
355 
356         r.map{stepResult => stepResult.step.name -> stepResult}.toMap
357       }
358     }
359     logInfo(s"Finished Flow id: ${flowId.id}, version: ${flowId.version} in ${timed._2.pretty}")
360 
361     FlowResult(timed._1, timed._2)
362   }
