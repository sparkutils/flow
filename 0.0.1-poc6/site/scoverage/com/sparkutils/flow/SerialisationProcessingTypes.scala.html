39     override def load(stepIds: Map[Id, StepRow], rsRows: Dataset[CombinedRuleSuiteRows], flowRuleGroup: FlowRuleGroup[Id]): Unit = {
40       val filtered =
41         if (flowRuleGroup.ruleGroup.isEmpty)
42           rsRows
43         else
44           rsRows.filter(
45             ShimUtils.callFunction("in", Seq(struct(col("ruleSuiteId"), col("ruleSuiteVersion"))) ++

47             )

51     }
52   }
