47   implicit val ofIds: FlowRuleGroupProcessing[Id] = new FlowRuleGroupProcessing[Id] {
48     override def load(stepIds: Map[Id, StepRow], rsRows: Dataset[CombinedRuleSuiteRows], flowRuleGroup: FlowRuleGroup[Id]): Unit = {
49       val filtered =
50         if (flowRuleGroup.ruleGroup.isEmpty)

52         else
53           rsRows.filter(
54             ShimUtils.callFunction("in", Seq(struct(col("ruleSuiteId"), col("ruleSuiteVersion"))) ++
55               flowRuleGroup.ruleGroup.map(id => struct(lit(id.id), lit(id.version))): _*
56             )

60     }
61   }
