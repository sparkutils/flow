19   /**
20    * Processes output fields for [[MergeFields]] and [[OutputFieldsOnly]]
21    *
22    * @param fieldName the field name to be used by the runner column
23    * @param runner the runnerColumn for interface completeness
24    * @param group_audit the group_audit column
25    * @param input the dataframe to add the output fields to
26    * @param outputFields a set of output fields which will be mapped to {{{s"$fieldName.result.$n"}}}
27    * @param extraFields any extra columns to be added
28    * @return
29    */
30   def withOutputFields(fieldName: String, runner: Column, group_audit: Column, input: DataFrame,
31                        outputFields: Set[String], extraFields: Set[Column] = Set.empty,
32                        useRunnerColumn: Boolean = false): DataFrame =
33     // cannot be run directly on dataFrame as otherwise an lca will be added to any repetitive expressions
34     // this stops row number plus another of other queries running correctly in all circumstances
35     input.select(
36       Seq(if (useRunnerColumn) runner else scol(fieldName), group_audit)
37         ++ extraFields ++
38         outputFields.map{n =>
39           scol(s"$fieldName.result.$n").as(n)
40         } :_*)
41 
42   /**
43    * Ueed by MergeFields and OutputFieldsOnly to add the runner, group_audit, and extra columns and the children of
44    * the runner.  This requires at least a single .schema action, but still only processes the remaining data in a single
45    * projection.
46    *
47    * In the MergeFields case extra is the original fields - the children of the runner and requires an additional .schema call.
48    *
49    * @return
50    */
51   def withoutOutputFields(fieldName: String, runner: Column, group_audit: Column, input: DataFrame,
52                           extra: Set[String] => Set[String] = _ => Set.empty): DataFrame = {
53 
54     val starter = input.select(runner)
55     val childrenNames = starter.schema.fields.head.dataType.asInstanceOf[StructType].
56       fields.find(_.name == "result").map(f => f.dataType.asInstanceOf[StructType].
57         fields.map(f => f.name).toSet).getOrElse(Set.empty[String])
58     val children = childrenNames.map(name => expr(s"$fieldName.result.${name}").as(name))
59 
60     input.select(Seq(runner, group_audit) ++ extra(childrenNames).map(scol).toSeq ++ children: _*)
61 
62   }

64   protected[flow] def resultProcessInputs[RP: RuleSuiteParam](step: Step[RP], runner: Column, childrenRaw: Seq[String],
65                                                               outputFieldsI: Option[Set[String]], ei: RunnerInputs,
66                                                               flow: FlowT[_,_]): ResultProcessInputs = {
67     import flow._
68 
69     val outputFields: Option[Set[String]] =
70       outputFieldsI.flatMap { s =>
71         if (step.options.boolean(forceMergeProjection))
72           None
73         else
74           Some(s)

77     val (starterColumns, existingAudit) =
78       // if flowAuditColName is present and the correct type, select all the others, assuming via having a parent step
79       // isn't enough if StarOnly or OutputFieldOnly is provided
80       //
81       (Seq(runner),
82         ei.hasAudit(flowAuditColName)(
83           Seq(scol(flowAuditColName))
84         )(
85           Seq.empty
86         )
87       )
88 
89     val fieldName = step.defaultFieldName
90 
91     val children = childrenRaw.map(n => scol(fieldName).getField(n).as(n))
92 
93     val group_auditF =
94       group_audit(scol(fieldName),
95         step.operation.combineAuditWith.map { fnames =>
96           fnames.map { fname => expr(fname) }.toSeq
97         }.getOrElse(Seq.empty) ++ existingAudit
98           : _*).as(flowAuditColName)
99 
100     ResultProcessInputs(outputFields, starterColumns, fieldName, children, group_auditF)
101   }

122                                                                   flow: FlowT[_,RP]): DataFrame = {
123     import flow._
124     import step.operation._
125     import step.options
126 
127     val RunnerOutput(runner, childrenRaw, outputFieldsI) = engine
128 
129     val ei@RunnerInputs(struct, withoutFlowAudit, _) = runnerInputs(options, dataFrame, step, flow)
130 
131     val ri@ResultProcessInputs(outputFields, starterColumns, fieldName, children, group_auditF) =
132       resultProcessInputs(step, runner, childrenRaw, outputFieldsI, ei, flow)
133 
134     resultApproach match {
135       case AsIs =>
136         dataFrame.select(starterColumns ++
137           ei.hasAudit(flowAuditColName)(
138             withoutFlowAudit.map(scol) ++ Seq(group_auditF)
139           )(
140             Seq(expr("*"), group_auditF)
141           ): _*)
142 
143       case MergeFields =>
144 
145         outputFields.fold {
146 
147           withoutOutputFields(fieldName, runner, group_auditF, dataFrame, extra = childrenNames => {
148             val og = dataFrame.columns.toSet
149             og -- childrenNames // only remove the result fields
150           })
151 
152         } { outputFields =>
153 
154           val fields = struct.map(_.name).toSet -- outputFields - flowAuditColName
155           withOutputFields(fieldName, runner, group_auditF, dataFrame, outputFields, extraFields = fields.map(scol),
156             useRunnerColumn = true)
157 
158         }
159 
160       case OutputFieldsOnly =>
161 
162         outputFields.fold {
163 
164           withoutOutputFields(fieldName, runner, group_auditF, dataFrame)
165 
166         } { o =>
167           withOutputFields(fieldName, runner, group_auditF, dataFrame, outputFields = o,
168             useRunnerColumn = true)
169         }
170 
171       case StarOnly => dataFrame.select(Seq(runner) ++ children: _*)
172       case OutputFieldOnly => dataFrame.select(runner)
173       case c: CustomApproach => c.customResultApproach(dataFrame, runner, ei, ri, step)
174     }
175   }
