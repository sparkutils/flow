22   /**
23    * Processes output fields for [[MergeFields]] and [[OutputFieldsOnly]]
24    *
25    * @param fieldName the field name to be used by the runner column
26    * @param runner the runnerColumn for interface completeness
27    * @param group_audit the group_audit column
28    * @param input the dataframe to add the output fields to
29    * @param outputFields a set of output fields which will be mapped to {{{s"$fieldName.result.$n"}}}
30    * @param extraFields any extra columns to be added
31    * @return
32    */
33   def withOutputFields(fieldName: String, runner: Column, group_audit: Column, input: DataFrame,
34                        outputFields: Set[String], extraFields: Set[Column] = Set.empty,
35                        useRunnerColumn: Boolean = false): DataFrame =
36     // cannot be run directly on dataFrame as otherwise an lca will be added to any repetitive expressions
37     // this stops row number plus another of other queries running correctly in all circumstances
38     input.select(
39       Seq(if (useRunnerColumn) runner else scol(fieldName), group_audit)
40         ++ extraFields ++
41         outputFields.map{n =>
42           scol(s"$fieldName.result.$n").as(n)
43         } :_*)
44 
45   /**
46    * Ueed by MergeFields and OutputFieldsOnly to add the runner, group_audit, and extra columns and the children of
47    * the runner.  This requires at least a single .schema action, but still only processes the remaining data in a single
48    * projection.
49    *
50    * In the MergeFields case extra is the original fields - the children of the runner and requires an additional .schema call.
51    *
52    * @return
53    */
54   def withoutOutputFields(fieldName: String, runner: Column, group_audit: Column, input: DataFrame,
55                           extra: Set[String] => Set[String] = _ => Set.empty): DataFrame = {
56 
57     val starter = input.select(runner)
58     val childrenNames = starter.schema.fields.head.dataType.asInstanceOf[StructType].
59       fields.find(_.name == "result").map(f => f.dataType.asInstanceOf[StructType].
60         fields.map(f => f.name).toSet).getOrElse(Set.empty[String])
61     val children = childrenNames.map(name => expr(s"$fieldName.result.${name}").as(name))
62 
63     input.select(Seq(runner, group_audit) ++ extra(childrenNames).map(scol).toSeq ++ children: _*)
64 
65   }

158           withOutputFields(fieldName, runner, group_auditF, dataFrame, outputFields, extraFields = fields.map(scol),

162 
163       case OutputFieldsOnly =>
164 
165         outputFields.fold {
166

172         }

175       case OutputFieldOnly => dataFrame.select(runner)
176       case c: CustomApproach => c.customResultApproach(dataFrame, runner, ei, ri, step)
177     }
178   }

180   /**
181    * When no stepTimeout has been configured for a Step, {{{promise.future}}} is called, otherwise failAfter is used.
182    * @param step
183    * @param promise
184    * @tparam T
185    * @tparam P
186    * @return
187    */
188   def stepTimeout[T, P](step: Step[T], promise: Promise[P])(implicit executionContext: ExecutionContext): Promise[P] = {
189     val stepTimeout = step.options.duration(stepTimeoutName, Duration.Inf)
190     if (stepTimeout == Duration.Inf)
191       promise
192     else
193       promise.failAfter(stepTimeout) {
194         StepException(step = step, timedOut = true)
195       }
196   }
