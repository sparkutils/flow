213    * Builds the chain of Promises from roots, completing roots triggers processing of the rest of the DAG

216   private def buildPromises(): (Map[String, Promise[StepResult[RP]]], Map[String, Promise[StepResult[RP]]]) = {
217     val rootPromises = roots.map(s => s.name -> Promise[StepResult[RP]]()).toMap
218     val promises = scala.collection.mutable.Map.empty ++ rootPromises
219 
220     def chaseDown(elem: theGraph.NodeT): Future[StepResult[RP]] = {
221       val cur = elem.source
222 
223       val r =
224         promises.get(cur.name).map(_.future).getOrElse {
225           val p = Promise[StepResult[RP]]()
226           promises.put(cur.name, p)
227 
228           // do their roots
229           val f = Future.sequence(
230             elem.diPredecessors.map {
231               parent =>
232                 chaseDown(parent)
233             }
234           )
235 
236           stepTimeout(cur, p)
237 
238           val newF: Future[StepResult[RP]] = f.map { names =>
239             performStep(names.head.output, cur, names)
240           }
241 
242           p.completeWith(newF)
243 
244           p.future
245         }
246 
247       // children should stitch on their parent only
248       elem.diSuccessors.foreach {
249         dependent =>
250           chaseDown(dependent)
251       }
252 
253       r
254     }
255     rest.foreach{r => chaseDown(theGraph.get(r))}
256 
257     (rootPromises, scala.collection.immutable.Map.empty ++ promises)
258   }

263   def verifySteps(): Unit = {
264     def badViewName(name: String): Boolean = {
265       name.isEmpty
266     }
267     try {
268       if (steps.isEmpty) {
269         throw FlowException(EmptyFlow)
270       }
271       if (steps.exists(s => (s.name eq null) || s.name.isEmpty)) {
272         throw FlowException(EmptyStepName)
273       }
274       steps.find(s => (s.data.inputView.exists(badViewName) || s.data.inputView.isEmpty) && s.dependencies.size > 1).foreach {
275         step =>
276           throw FlowException(DefaultViewNamesMultipleParents(step))
277       }
278       steps.find(s => s.data.inputView.exists(badViewName) || s.data.outputView.exists(badViewName)).foreach {
279         step =>
280           throw FlowException(InvalidViewNames(step))
281       }
282       val dupes = steps.groupBy(_.name).filter(_._2.size > 1)
283       if (dupes.nonEmpty) {
284         throw FlowException(s"$DuplicateNames (${dupes.keys.mkString(",")})")
285       }
286       val cyc = theGraph.findCycle
287       // for info only val paths = roots.map( n => n -> theGraph.get(n).outerNodeTraverser.toSeq )
288       cyc.foreach {
289         cycle =>
290           throw FlowException(s"$CycleDetected $cycle")
291       }
292 
293       steps.foreach {
294         step =>
295           (step.operation.function, step.operation.resultApproach) match {
296             case (DQ, MergeFields | OutputFieldsOnly) =>
297               throw FlowException(InvalidDQResultApproach(step))
298             case _ => ()
299           }
300       }
301     } catch {
302       case f: FlowException =>
303         logError(f.msg, f)
304         throw f
305     }

309    * A convenience function to start processing with an optionally provided DataFrame registered as the first steps inputView.
310    *
311    * The entire Flow must be complete within the Flow.duration
312    *
313    * @param sparkSession
314    * @param starting a function called for each Root Step (a Step having no dependencies), when a DataFrame is provided
315    *                 it is registered as a temporary view with the name of the inputView.  When None is returned and the
316    *                 inputView is not present an error is thrown, this is delegated to the FlowDataHandling.loadData function
317    * @param tolerant defaulting to false, an exception in an Step triggers an Exception from this function, a successful
318    *                 result will only have [[StepResult]]s.  Providing true will capture both successes and failures
319    * @return a map of step name to tuples of end Steps with their resulting DataFrames and run statistics,
320    *         note although the Step returned may have been replaced by modifyStep
321    *         the name key will be the original [[Step.name]]

323   def run(sparkSession: SparkSession, starting: Step[RP] => Option[DataFrame], tolerant: Boolean = false): FlowResult[RP] = {
324     val timed = Timer {
325       verifySteps()
326       logInfo(s"Starting Flow id: ${flowId.id}, version: ${flowId.version}")
327 
328       val (rootPromises, promises) = buildPromises()
329 
330       // processing th roots sets of the rest
331       roots.par.map {
332         root =>
333           val p = rootPromises(root.name)
334           try {
335             val df = starting(root)
336             val token = rootToken(root)
337 
338             stepTimeout(root, p)
339 
340             val r =
341               df.fold(
342                 performStep(loadData(sparkSession = sparkSession, token = token), root, Set.empty)
343               ) { df =>
344                 df.createOrReplaceTempView(token)
345                 performStep(df, root, Set.empty)
346               }
347 
348             p success r
349           } catch {
350             case t: Throwable => p tryFailure StepException(root, t) // wrap this in an exception with the root
351           }
352       }
353 
354       if (tolerant) {
355         val all = Future.sequence((rootPromises ++ promises).map(_._2.future.transform(Success(_))))
356         val r = Await.result(all, duration)
357 
358         r.map {
359           case Success(stepResult: StepResult[RP]) => stepResult.step.name -> stepResult
360           case Failure(stepException: StepException[RP]) => stepException.step.name -> stepException
361         }.toMap
362       } else {
363         val all = Future.sequence((rootPromises ++ promises).map(_._2.future))
364         val r = Await.result(all, duration)
365 
366         r.map{stepResult => stepResult.step.name -> stepResult}.toMap
367       }
368     }
369     logInfo(s"Finished Flow id: ${flowId.id}, version: ${flowId.version} in ${timed._2.pretty}")
370 
371     FlowResult(timed._1, timed._2)
372   }
